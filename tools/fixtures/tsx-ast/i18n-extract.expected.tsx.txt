import { useState } from 'react';
import { addToast } from '../store';
import { useTranslation } from 'react-i18next';
import { k } from '../i18n';

const TABLE = { label: k("Module level label"), other: ('Wrapped module label') };

export function Fx({ count, label = 'Default label' }: { count: number; label?: string }) {
    const { t } = useTranslation();
    const { title: heading = 'Untitled draft' } = { title: undefined as string | undefined };
    const [open] = useState(false);
    const options = { label: ('Wrapped option'), hint: t("Plain hint") };
    const save = () => addToast('success', ('Saved the course'));
    return (
        <section title={t("Favourite things")} onClick={save}>
            <h2>
                {t("Mark done anyway")}
            </h2>
            <p>{t("{{count}} cards due", { count: (count) })}</p>
            <p>
                {t("Showing {{count}} results", { count })}
            </p>
            <pre>
                Sample output here
            </pre>
            <p>{open ? (t("Opened panel")) : t("Closed panel")}</p>
            <input placeholder={t("Type a course name")} aria-label={t("Search all courses")} />
            <em>{t("{{count}} topics left", { count })}</em>
            {heading}{label}{options.label}{TABLE.other}
        </section>
    );
}
